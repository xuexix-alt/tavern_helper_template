<%
// 动态注入角色档案（@INJECT 专用）：
// - 约束：@INJECT 条目本身应保持“未启用”，由提示词模板语法插件在生成提示词时执行此 EJS。
// - 兼容：避免使用 optional chaining / 不依赖“列出已启用世界书条目”的 API（不同环境可能不存在）。
//
// 策略（宽容触发，提及/登场都可）：
// - 提及：本轮用户输入 + 上轮AI输出 里点名/别名命中 -> 注入对应角色档案。
// - 登场：stat_data 中登场 -> 也注入对应角色档案。
// - 动态新角色：若角色档案条目命名不统一，但关键字里包含角色名，则尝试用
//   getEnabledWorldInfoEntries()（若存在）按关键字命中并用 uid 注入（不依赖标题规范）。
//
// 如果你更想“仅当提及才注入”，可以再加一层关键词命中；但为了保证不漏注入，这里优先稳妥策略。

const stat = getvar('stat_data', { defaults: {} });
function isRecord(v) {
  return !!(v && typeof v === 'object' && !Array.isArray(v));
}

if (!isRecord(stat)) return;

const reserved = {
  '世界': true,
  '主线任务': true,
  '庇护所': true,
  '房间': true,
  '楼层其他住户': true,
  '临时NPC': true,
  '_变量': true,
};

function isOnstageRole(v) {
  return isRecord(v) && v['登场状态'] === '登场';
}

function isDeadRole(v) {
  if (!isRecord(v)) return false;
  const health = Number(v['健康']);
  const imp = Number(v['秩序刻印']);
  if (Number.isFinite(health) && health <= 0) return true;
  if (Number.isFinite(imp) && imp < 0) return true;
  return false;
}

function safeString(v) {
  return v === null || v === undefined ? '' : String(v);
}

function getLastByRole(role) {
  try {
    if (role === 'user' && typeof lastUserMessageId === 'number') return safeString(getChatMessage(lastUserMessageId, 'user'));
    if (role === 'assistant' && typeof lastCharMessageId === 'number') return safeString(getChatMessage(lastCharMessageId, 'assistant'));
  } catch (e) {}

  try {
    const arr = getChatMessages(1, role);
    if (Array.isArray(arr) && arr.length > 0) return safeString(arr[arr.length - 1]);
  } catch (e) {}

  return '';
}

const mentionText = (getLastByRole('user') + '\n' + getLastByRole('assistant')).trim();

function includesAny(text, list) {
  if (!text) return false;
  if (!Array.isArray(list)) return false;
  for (let i = 0; i < list.length; i++) {
    const kw = String(list[i] || '').trim();
    // 过滤过短别名，避免误触发。
    if (!kw || kw.length < 2) continue;
    if (text.indexOf(kw) !== -1) return true;
  }
  return false;
}

function parseDossierIndex(text) {
  try {
    const data = JSON.parse(String(text || ''));
    if (!Array.isArray(data)) return [];
    const out = [];
    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      if (!row || typeof row !== 'object') continue;
      const name = String(row.name || '').trim();
      if (!name) continue;
      const aliases = Array.isArray(row.aliases) ? row.aliases : [];
      out.push({ name, aliases });
    }
    return out;
  } catch (e) {
    return [];
  }
}

const MAX_INJECT = 3;

function uniqPushByKey(list, key, item) {
  for (let i = 0; i < list.length; i++) {
    if (list[i] && list[i].key === key) return;
  }
  list.push(item);
}

// 0) 动态新角色（若可用）：按关键字命中“已启用”的世界书条目，然后按 uid 注入
const hitEntries = [];
if (mentionText && typeof getEnabledWorldInfoEntries === 'function') {
  try {
    const allEntries = await getEnabledWorldInfoEntries(true, true, true, true, true);
    const arr = Array.isArray(allEntries) ? allEntries : [];
    for (let i = 0; i < arr.length; i++) {
      const e = arr[i];
      const comment = String((e && e.comment) || '').trim();
      if (!comment || comment.indexOf('角色档案') === -1) continue;
      const keys = []
        .concat(Array.isArray(e && e.key) ? e.key : [])
        .concat(Array.isArray(e && e.keysecondary) ? e.keysecondary : []);
      if (!includesAny(mentionText, keys)) continue;
      const uid = e && e.uid;
      if (uid === null || uid === undefined) continue;
      uniqPushByKey(hitEntries, 'uid:' + String(uid), { key: 'uid:' + String(uid), uid: Number(uid), label: comment });
      if (hitEntries.length >= MAX_INJECT) break;
    }
  } catch (e) {}
}

// 1) 提及：按“角色档案索引”命中（不依赖世界书启用状态）
const mentionTargets = [];
if (mentionText) {
  const indexText = await getwi('角色档案索引');
  const index = parseDossierIndex(indexText);
  for (let i = 0; i < index.length; i++) {
    const name = index[i].name;
    const aliases = index[i].aliases;
    const hit = mentionText.indexOf(name) !== -1 || includesAny(mentionText, aliases);
    if (!hit) continue;
    uniqPushByKey(mentionTargets, 'role:' + name, { key: 'role:' + name, roleName: name, label: '角色档案 - ' + name });
    if (mentionTargets.length >= MAX_INJECT) break;
  }
}

// 2) 登场：按“登场变量”挑选（兜底，不依赖本轮提及）
const onstageTargets = [];
for (const k in stat) {
  if (!Object.prototype.hasOwnProperty.call(stat, k)) continue;
  if (!k || reserved[k] || k.charAt(0) === '_') continue;
  const v = stat[k];
  if (!isOnstageRole(v) || isDeadRole(v)) continue;
  const roleName = String(k).trim();
  if (!roleName) continue;
  uniqPushByKey(onstageTargets, 'role:' + roleName, { key: 'role:' + roleName, roleName, label: '角色档案 - ' + roleName });
}

const tempNpc = stat['临时NPC'];
if (isRecord(tempNpc)) {
  for (const k in tempNpc) {
    if (!Object.prototype.hasOwnProperty.call(tempNpc, k)) continue;
    if (!k || k.charAt(0) === '_') continue;
    const v = tempNpc[k];
    if (!isOnstageRole(v) || isDeadRole(v)) continue;
    const roleName = String(k).trim();
    if (!roleName) continue;
    uniqPushByKey(onstageTargets, 'role:' + roleName, { key: 'role:' + roleName, roleName, label: '角色档案 - ' + roleName });
  }
}

// 合并：动态命中 uid 优先，其次提及命中角色名，其次登场角色名；总量限制 MAX_INJECT
const finalTargets = [];
for (let i = 0; i < hitEntries.length; i++) {
  if (finalTargets.length >= MAX_INJECT) break;
  finalTargets.push(hitEntries[i]);
}
for (let i = 0; i < mentionTargets.length; i++) {
  if (finalTargets.length >= MAX_INJECT) break;
  uniqPushByKey(finalTargets, mentionTargets[i].key, mentionTargets[i]);
}
// 登场目标按名字稳定排序，避免每轮抖动
onstageTargets.sort((a, b) => (a.label < b.label ? -1 : a.label > b.label ? 1 : 0));
for (let i = 0; i < onstageTargets.length; i++) {
  if (finalTargets.length >= MAX_INJECT) break;
  uniqPushByKey(finalTargets, onstageTargets[i].key, onstageTargets[i]);
}

if (finalTargets.length === 0) return;

print('<角色档案_动态注入>\\n');
for (let i = 0; i < finalTargets.length; i++) {
  const t = finalTargets[i];

  // uid 注入：用于“命名不统一但关键字命中”的动态角色档案
  if (t && t.uid !== undefined && t.uid !== null) {
    const wi = await getwi(Number(t.uid));
    if (!wi) continue;
    const label = String(t.label || '角色档案').trim() || '角色档案';
    print('\\n<' + label + '>\\n');
    print(String(wi).trimEnd());
    print('\\n</' + label + '>\\n');
    continue;
  }

  // roleName 注入：要求条目标题可用 `角色档案 - <姓名>` 读取
  const roleName = t && t.roleName ? String(t.roleName).trim() : '';
  if (!roleName) continue;
  const title = '角色档案 - ' + roleName;
  const wi = await getwi(title);
  if (!wi) continue;
  print('\\n<' + title + '>\\n');
  print(String(wi).trimEnd());
  print('\\n</' + title + '>\\n');
}
print('\\n</角色档案_动态注入>');
%>
