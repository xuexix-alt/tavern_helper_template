function isRecord(v) {
  return !!(v && typeof v === 'object' && !Array.isArray(v));
}

function toBool(v) {
  if (v === true || v === false) return v;
  if (typeof v === 'number') return v !== 0;
  const s = String(v === null || v === undefined ? '' : v)
    .trim()
    .toLowerCase();
  return s === '1' || s === 'true' || s === 'yes' || s === 'on' || s === '开启';
}

function readInjectDebugEnabled() {

  for (let i = 0; i < candidates.length; i++) {
    const v = candidates[i];
    if (v === null || v === undefined || v === '') continue;

  }
  return false;
}

function normalizeRoleList(list) {
  if (!Array.isArray(list)) return [];
  const seen = {};
  const out = [];
  for (let i = 0; i < list.length; i++) {
    const name = String(list[i] || '').trim();
    if (!name || seen[name]) continue;
    seen[name] = true;
    out.push(name);
  }
  return out;
}

function isDeadRole(v) {
  if (!isRecord(v)) return false;
  const health = Number(v['健康']);
  const imp = Number(v['秩序刻印']);
  if (Number.isFinite(health) && health <= 0) return true;
  if (Number.isFinite(imp) && imp < 0) return true;
  return false;
}

function safeString(v) {
  return v === null || v === undefined ? '' : String(v);
}

const LOOKBACK_FLOORS = 4;

function extractMessageText(v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'string') return v;
  if (typeof v === 'number' || typeof v === 'boolean') return String(v);
  if (typeof v !== 'object') return String(v);

  if (typeof v.message === 'string') return v.message;
  if (typeof v.content === 'string') return v.content;
  if (typeof v.text === 'string') return v.text;

  if (Array.isArray(v.swipes) && v.swipes.length > 0) {
    const swipeId = Number(v.swipe_id);
    if (Number.isFinite(swipeId) && swipeId >= 0 && swipeId < v.swipes.length) {
      return safeString(v.swipes[swipeId]);
    }
    return safeString(v.swipes[v.swipes.length - 1]);
  }

  return safeString(v);
}

function getRecentByRole(role) {

        const begin = Math.max(0, latest - LOOKBACK_FLOORS);
        const arr = getChatMessages(String(begin) + '-' + String(latest), { role });
        if (Array.isArray(arr) && arr.length > 0) {
          const texts = [];
          for (let i = 0; i < arr.length; i++) {
            const t = extractMessageText(arr[i]).trim();
            if (t) texts.push(t);
          }
          if (texts.length > 0) return texts.join('\n');
        }
      }
    }
  } catch (e) {}

  // 兼容兜底：某些环境 getChatMessages 仍是旧签名 (count, role)。
  try {
    const arr = getChatMessages(LOOKBACK_FLOORS + 1, role);
    if (Array.isArray(arr) && arr.length > 0) {
      const texts = [];
      for (let i = 0; i < arr.length; i++) {
        const t = extractMessageText(arr[i]).trim();
        if (t) texts.push(t);
      }
      if (texts.length > 0) return texts.slice(-(LOOKBACK_FLOORS + 1)).join('\n');
    }
  } catch (e) {}

  // 最后兜底：只取 last*MessageId 的单条，避免为空。
  try {
    if (role === 'user' && typeof lastUserMessageId === 'number') return extractMessageText(getChatMessage(lastUserMessageId, 'user'));
    if (role === 'assistant' && typeof lastCharMessageId === 'number') return extractMessageText(getChatMessage(lastCharMessageId, 'assistant'));
    if (role === 'system' && typeof lastCharMessageId === 'number') return extractMessageText(getChatMessage(lastCharMessageId, 'system'));
  } catch (e) {}

  return '';
}

function includesAny(text, list) {
  if (!text) return false;
  if (!Array.isArray(list)) return false;
  for (let i = 0; i < list.length; i++) {
    const kw = String(list[i] || '').trim();
    // 过滤过短别名，避免误触发。
    if (!kw || kw.length < 2) continue;
    if (text.indexOf(kw) !== -1) return true;
  }
  return false;
}

function parseDossierIndex(text) {
  try {
    const data = JSON.parse(String(text || ''));
    if (!Array.isArray(data)) return [];
    const out = [];
    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      if (!row || typeof row !== 'object') continue;
      const name = String(row.name || '').trim();
      if (!name) continue;
      const aliases = Array.isArray(row.aliases) ? row.aliases : [];
      const title = typeof row.title === 'string' ? String(row.title).trim() : '';
      out.push({ name, aliases, title });
    }
    return out;
  } catch (e) {
    return [];
  }
}

function extractRoleNameFromComment(comment) {
  const text = String(comment || '').trim();
  if (!text) return '';
  const match = text.match(/角色档案\\s*[-—–:：]?\\s*(.+)$/);
  if (!match) return '';
  return String(match[1] || '').trim();
}

function uniqPushByKey(list, key, item) {
  for (let i = 0; i < list.length; i++) {
    if (list[i] && list[i].key === key) return;
  }
  list.push(item);
}

function buildRoleAllowSet(list) {
  const normalized = normalizeRoleList(list);
  if (!normalized.length) return null;
  const set = {};
  for (let i = 0; i < normalized.length; i++) {
    const k = String(normalized[i] || '').trim();
    if (roleSelectorDeletedSet && roleSelectorDeletedSet[k] === true) continue;
    if (k) set[k] = true;
  }
  return set;
}

const roleSelectorAllowSet = (() => {
  // 严格模式：初始化后仅 selected 生效，revealed 仅代表“可见可选”。
  if (roleSelectorStrictMode) return buildRoleAllowSet(roleSelectorSelected) || {};

  // 兼容模式：旧存档/未初始化时，允许 selected + revealed 并集。
  const all = []
    .concat(Array.isArray(roleSelectorSelected) ? roleSelectorSelected : [])
    .concat(Array.isArray(roleSelectorRevealed) ? roleSelectorRevealed : []);
  return buildRoleAllowSet(all);
})();

function isRoleAllowedBySelector(roleName) {
  const name = String(roleName || '').trim();
  if (!name) return false;
  if (roleSelectorDeletedSet && roleSelectorDeletedSet[name] === true) return false;
  // 未初始化选择器时不做限制，保持兼容历史存档。
  if (!roleSelectorAllowSet) return true;
  return roleSelectorAllowSet[name] === true;
}

function buildRoleDossierTitleCandidates(roleName) {
  const name = String(roleName || '').trim();
  if (!name) return [];

  const row = dossierIndexMap[name] || null;
  const nameVariants = [];
  const pushName = v => {
    const x = String(v || '').trim();
    if (!x) return;
    if (nameVariants.indexOf(x) === -1) nameVariants.push(x);
  };
  pushName(name);
  pushName(name.replace(/\?/g, '・'));
  pushName(name.replace(/・/g, '?'));
  pushName(name.replace(/·/g, '・'));
  pushName(name.replace(/•/g, '・'));

  const titles = [];
  const pushTitle = v => {
    const x = String(v || '').trim();
    if (!x) return;
    if (titles.indexOf(x) === -1) titles.push(x);
  };

  if (row && typeof row.title === 'string' && row.title.trim()) {
    pushTitle(row.title.trim());
  }

  for (let i = 0; i < nameVariants.length; i++) {
    const n = nameVariants[i];
    pushTitle('角色档案 - ' + n);
    pushTitle('角色档案- ' + n);
    pushTitle('角色档案-' + n);
    pushTitle('角色档案：' + n);
    pushTitle('角色档案_' + n);
    pushTitle('角色档案_-_' + n);
  }

  return titles;
}

async function getRoleDossierByName(roleName) {
  const name = String(roleName || '').trim();
  if (!name) return null;

  const candidates = buildRoleDossierTitleCandidates(name);
  for (let i = 0; i < candidates.length; i++) {
    const title = candidates[i];
    try {
      const wi = await getwi(title);
      if (wi) return { label: title, content: wi };
    } catch (e) {}
  }

  // 宽容兜底：若环境提供“已启用条目”列表，则尝试按注释/关键字命中后用 uid 读取。
  if (typeof getEnabledWorldInfoEntries === 'function') {
    try {
      const allEntries = await getEnabledWorldInfoEntries(true, true, true, true, true);
      const arr = Array.isArray(allEntries) ? allEntries : [];
      for (let i = 0; i < arr.length; i++) {
        const e = arr[i];
        const comment = String((e && e.comment) || '').trim();
        const keys = []
          .concat(Array.isArray(e && e.key) ? e.key : [])
          .concat(Array.isArray(e && e.keysecondary) ? e.keysecondary : []);
        const parsedRoleName = extractRoleNameFromComment(comment);
        const hitByComment = !!parsedRoleName && parsedRoleName.indexOf(name) !== -1;
        const hitByKeys = includesAny(name, keys) || includesAny(comment, [name]);
        if (!(hitByComment || hitByKeys)) continue;
        const uid = e && e.uid;
        if (uid === null || uid === undefined) continue;
        const wi = await getwi(Number(uid));
        if (wi) return { label: comment || ('角色档案 - ' + name), content: wi };
      }
    } catch (e) {}
  }

  return null;
}

  try {
    const allEntries = await getEnabledWorldInfoEntries(true, true, true, true, true);
    const arr = Array.isArray(allEntries) ? allEntries : [];
    for (let i = 0; i < arr.length; i++) {
      const e = arr[i];
      const comment = String((e && e.comment) || '').trim();
      if (!comment || comment.indexOf('角色档案') === -1) continue;
      const parsedRoleName = extractRoleNameFromComment(comment);
      if (parsedRoleName && !isRoleAllowedBySelector(parsedRoleName)) continue;
      const keys = []
        .concat(Array.isArray(e && e.key) ? e.key : [])
        .concat(Array.isArray(e && e.keysecondary) ? e.keysecondary : []);
      if (!includesAny(mentionText, keys)) continue;
      const uid = e && e.uid;
      if (uid === null || uid === undefined) continue;
      uniqPushByKey(hitEntries, 'uid:' + String(uid), { key: 'uid:' + String(uid), uid: Number(uid), label: comment });
    }
  } catch (e) {}

// 2) 登场：按“登场变量”挑选（兜底，不依赖本轮提及）

  if (!isOnstageRole(v) || isDeadRole(v)) continue;

  if (!roleName) continue;

function debugRoleLabel(t) {
  if (t && t.roleName) return String(t.roleName).trim();
  if (t && t.label) return String(t.label).trim();
  if (t && t.uid !== undefined && t.uid !== null) return 'uid:' + String(t.uid);
  return '';
}

function printInjectDebugBlock(finalTargetsList) {
  if (!injectDebugEnabled) return;

  const mentionNames = mentionTargets.map(debugRoleLabel).filter(Boolean);
  const onstageNames = onstageTargets.map(debugRoleLabel).filter(Boolean);
  const selectorNames = selectorTargets.map(debugRoleLabel).filter(Boolean);
  const selectorAllowNames = roleSelectorAllowSet ? Object.keys(roleSelectorAllowSet).sort() : [];
  const selectorDeletedNames = roleSelectorDeletedSet ? Object.keys(roleSelectorDeletedSet).sort() : [];
  const uidNames = hitEntries.map(debugRoleLabel).filter(Boolean);
  const finalNames = (finalTargetsList || []).map(debugRoleLabel).filter(Boolean);
  const textPreview = mentionText
    ? mentionText
        .replace(/\s+/g, ' ')
        .slice(0, 120)
    : '（空）';

  print('\n<角色档案_动态注入_调试>\n');
  print('开关: 开启\n');
  print('选择器模式: ' + (roleSelectorStrictMode ? '严格（仅selected生效）' : '兼容（selected/revealed）') + '\n');
  print('提及命中: ' + (mentionNames.length > 0 ? mentionNames.join('、') : '无') + '\n');
  print('登场命中: ' + (onstageNames.length > 0 ? onstageNames.join('、') : '无') + '\n');
  print('选择器白名单: ' + (selectorAllowNames.length > 0 ? selectorAllowNames.join('、') : '未启用') + '\n');
  print('选择器已删除: ' + (selectorDeletedNames.length > 0 ? selectorDeletedNames.join('、') : '无') + '\n');
